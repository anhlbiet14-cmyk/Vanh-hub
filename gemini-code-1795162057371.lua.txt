local function detectSea()
    local pid = game.PlaceId
    if pid == 2753915549 then return 1 end
    if pid == 4442272183 then return 2 end
    if pid == 7449423635 then return 3 end

    local map = workspace:FindFirstChild("Map")
    if map then
        if map:FindFirstChild("Turtle") or map:FindFirstChild("PortTown") or map:FindFirstChild("CakeLoaf") then
            return 3
        end
        if map:FindFirstChild("Dressrosa") or map:FindFirstChild("Kingdom of Rose") or map:FindFirstChild("GreenBit") then
            return 2
        end
    end

    local ok, lv = pcall(function() return LP.Data.Level.Value end)
    if ok and tonumber(lv) then
        if lv >= 1500 then return 3 end
        if lv >= 700 then return 2 end
    end
    return 1
end
local CurrentSea = detectSea()

local function connect(signal, fn)
    local c = signal:Connect(fn)
    table.insert(State.connections, c)
    return c
end

local function safe(fn, ...)
    local ok, err = pcall(fn, ...)
    if not ok then warn("[Vanh Hub] " .. tostring(err)) end
end

local function getChar()
    local char = LP.Character
    if not char then return end
    return char,
        char:FindFirstChildOfClass("Humanoid"),
        char:FindFirstChild("HumanoidRootPart")
end

local function isPointer(input)
    return input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch
end

local function getRemote(name)
    local remotes = Svc.ReplicatedStorage:FindFirstChild("Remotes")
    return remotes and remotes:FindFirstChild(name) or nil
end

local function invokeCommF(...)
    local comm = getRemote("CommF_")
    if not comm then return nil end
    local ok, res = pcall(comm.InvokeServer, comm, ...)
    return ok and res or nil
end

local function playerNumber(name)
    local ok, v = pcall(function() return LP.Data[name].Value end)
    if ok and tonumber(v) then return tonumber(v) end
    local ok2, v2 = pcall(function() return LP.leaderstats[name].Value end)
    return ok2 and tonumber(v2) or nil
end

local function restoreNoclip()
    for part in pairs(State.noclipped) do
        if part and part.Parent then part.CanCollide = true end
    end
    table.clear(State.noclipped)
end

local function stopFarmBody()
    if State.farmBV then
        State.farmBV:Destroy()
        State.farmBV = nil
    end
end

local function anyAutoActive()
    return config.autoFarmLevel or config.autoFarmNear or config.autoFarmAllBoss
        or config.autoMastery or config.autoChest or config.autoFarmRaid
        or config.autoTrialV4 or config.autoMirageGear
        or (config.huntOnly and config.huntName ~= nil)
end

local UI = {}

function UI.new(class, props, parent)
    local o = Instance.new(class)
    for k, v in pairs(props or {}) do o[k] = v end
    if parent then o.Parent = parent end
    return o
end

function UI.corner(obj, r) return UI.new("UICorner", { CornerRadius = UDim.new(0, r) }, obj) end

function UI.stroke(obj, color, thickness, transparency)
    return UI.new("UIStroke", {
        Color = color, Thickness = thickness or 1, Transparency = transparency or 0,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
    }, obj)
end

function UI.grad(obj, c1, c2, rotation)
    obj.BackgroundColor3 = Color3.new(1, 1, 1)
    return UI.new("UIGradient", { Color = ColorSequence.new(c1, c2), Rotation = rotation or 0 }, obj)
end

function UI.tween(obj, props, t)
    Svc.TweenService:Create(obj,
        TweenInfo.new(t or 0.15, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), props):Play()
end

function UI.text(parent, str, size, color, font, props)
    local l = UI.new("TextLabel", {
        BackgroundTransparency = 1, BorderSizePixel = 0, Text = str, TextSize = size or 13,
        TextColor3 = color or T.text, Font = font or Enum.Font.GothamMedium,
        TextXAlignment = Enum.TextXAlignment.Left, TextTruncate = Enum.TextTruncate.AtEnd,
    })
    for k, v in pairs(props or {}) do l[k] = v end
    l.Parent = parent
    return l
end

local function beginDrag(d)
    if State.activeDrag and State.activeDrag.finish then State.activeDrag.finish() end
    State.activeDrag = d
end

connect(Svc.UIS.InputChanged, function(input)
    if State.activeDrag and (input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch) then
        State.activeDrag.move(input)
    end
end)
connect(Svc.UIS.InputEnded, function(input)
    if State.activeDrag and isPointer(input) then
        local d = State.activeDrag
        State.activeDrag = nil
        if d.finish then d.finish() end
    end
end)

local function makeDraggable(handle, dragTarget, onTap)
    handle.InputBegan:Connect(function(input)
        if not isPointer(input) then return end
        local startPointer = input.Position
        local startPos = dragTarget.Position
        local moved = false
        beginDrag({
            move = function(i)
                local d = i.Position - startPointer
                if not moved and d.Magnitude > 5 then moved = true end
                if moved then
                    dragTarget.Position = UDim2.new(
                        startPos.X.Scale, startPos.X.Offset + d.X,
                        startPos.Y.Scale, startPos.Y.Offset + d.Y)
                end
            end,
            finish = function()
                if not moved and onTap then onTap() end
            end,
        })
    end)
end

local function slider(page, label, key, min, max, step, cb, unit)
    local row = newRow(page, 54)
    UI.text(row, label, 13, T.text, nil, { Size = UDim2.new(0.65, 0, 0, 24), Position = UDim2.new(0, 12, 0, 4) })
    local valueLbl = UI.text(row, "", 12, T.accent, Enum.Font.GothamBold, {
        Size = UDim2.new(0.35, -12, 0, 24), Position = UDim2.new(0.65, 0, 0, 4),
        TextXAlignment = Enum.TextXAlignment.Right,
    })
    local track = UI.new("Frame", {
        Size = UDim2.new(1, -24, 0, 6), Position = UDim2.new(0, 12, 0, 36),
        BackgroundColor3 = T.off, BorderSizePixel = 0,
    }, row)
    UI.corner(track, 3)
    local fill = UI.new("Frame", { Size = UDim2.new(0, 0, 1, 0), BorderSizePixel = 0 }, track)
    UI.corner(fill, 3)
    UI.grad(fill, T.accent, T.accent2, 0)
    local knob = UI.new("Frame", {
        Size = UDim2.new(0, 14, 0, 14), AnchorPoint = Vector2.new(0.5, 0.5),
        Position = UDim2.new(0, 0, 0.5, 0), BackgroundColor3 = Color3.new(1, 1, 1), BorderSizePixel = 0,
    }, track)
    UI.corner(knob, 7)
    local hit = UI.new("TextButton", {
        Size = UDim2.new(1, 0, 0, 26), Position = UDim2.new(0, 0, 0, 28),
        BackgroundTransparency = 1, Text = "", BorderSizePixel = 0,
    }, row)

    local function render()
        local frac = (config[key] - min) / (max - min)
        fill.Size = UDim2.new(frac, 0, 1, 0)
        knob.Position = UDim2.new(frac, 0, 0.5, 0)
        valueLbl.Text = tostring(config[key]) .. (unit or "")
    end
    local function setFromX(x)
        local w = track.AbsoluteSize.X
        if w <= 0 then return end
        local frac = math.clamp((x - track.AbsolutePosition.X) / w, 0, 1)
        local v = math.clamp(math.floor((min + frac * (max - min)) / step + 0.5) * step, min, max)
        if v ~= config[key] then
            config[key] = v
            render()
            if cb then safe(cb, v) end
        end
    end
    hit.InputBegan:Connect(function(input)
        if not isPointer(input) then return end

local function teleportTo(cf)
    local _, _, root = getChar()
    if not root then return 0 end
    if State.currentTween then State.currentTween:Cancel() State.currentTween = nil end
    if not config.smoothTp then
        root.CFrame = cf
        return 0
    end
    local dist = (root.Position - cf.Position).Magnitude
    local dur = math.max(dist / config.tpSpeed, 0.05)
    State.currentTween = Svc.TweenService:Create(root, TweenInfo.new(dur, Enum.EasingStyle.Linear), { CFrame = cf })
    State.currentTween:Play()
    return dur
end

local function moveTowards(root, destPos, lookPos, dt)
    local diff = destPos - root.Position
    local dist = diff.Magnitude
    local goal
    if (destPos - lookPos).Magnitude < 0.1 or (math.abs(destPos.X - lookPos.X) < 0.05 and math.abs(destPos.Z - lookPos.Z) < 0.05) then
        goal = CFrame.new(destPos) * CFrame.Angles(math.rad(-90), 0, 0)
    else
        goal = CFrame.lookAt(destPos, lookPos)
    end

    if config.smoothTp and dist > 25 then
        local step = math.min(dist, config.tpSpeed * dt)
        local nextPos = root.Position + diff.Unit * step
        local flatLook = Vector3.new(lookPos.X, nextPos.Y, lookPos.Z)
        if (flatLook - nextPos).Magnitude > 0.1 then
            root.CFrame = CFrame.lookAt(nextPos, flatLook)
        else
            root.CFrame = CFrame.new(nextPos)
        end
    else
        root.CFrame = goal
    end
    root.AssemblyLinearVelocity = Vector3.zero
    root.AssemblyAngularVelocity = Vector3.zero
end

local function findPart(obj)
    if obj:IsA("BasePart") then return obj end
    return obj:FindFirstChild("Handle") or obj:FindFirstChildWhichIsA("BasePart", true)
end

local function isFruit(obj)
    if not obj.Name:find("Fruit") then return false end
    if not (obj:IsA("Tool") or obj:IsA("Model") or obj:IsA("BasePart")) then return false end
    if Svc.Players:GetPlayerFromCharacter(obj) then return false end
    if obj:FindFirstChildOfClass("Humanoid") then return false end
    return true
end

local function getFruits()
    local list = {}
    for _, obj in ipairs(workspace:GetChildren()) do
        if isFruit(obj) then
            local part = findPart(obj)
            if part then list[#list + 1] = { obj = obj, part = part, name = obj.Name } end
        end
    end
    return list
end

local function getChests()
    local list = {}
    local function scan(container)
        for _, obj in ipairs(container:GetChildren()) do
            if obj.Name:find("Chest") and not obj:FindFirstChildOfClass("Humanoid") then
                local part = findPart(obj)
                if part then list[#list + 1] = { obj = obj, part = part, name = obj.Name } end
            end
        end
    end
    scan(workspace)
    local folder = workspace:FindFirstChild("ChestModels")
    if folder then scan(folder) end
    return list
end

local function isAlive(m)
    if not (m and m.Parent) then return false end
    local hum = m:FindFirstChildOfClass("Humanoid")
    local hrp = m:FindFirstChild("HumanoidRootPart")
    if not (hum and hrp and hum.Health > 0) then return false end
    -- Bỏ qua quái rơi xuống đáy biển
    if hrp.Position.Y < -10 then return false end
    return true
end

-- Quét Boss cả trong workspace.Enemies và ReplicatedStorage
local function getActiveBosses()
    local list, seen = {}, {}
    local function checkFolder(folder, inWorkspace)
        if not folder then return end
        for _, m in ipairs(folder:GetChildren()) do
            if not seen[m.Name] then
                for bName, info in pairs(BossQuests) do
                    if m.Name == bName or m.Name:find(bName, 1, true) then
                        local hum = m:FindFirstChildOfClass("Humanoid")
                        local hrp = m:FindFirstChild("HumanoidRootPart")
                        if hum and hrp and hum.Health > 0 then
                            seen[m.Name] = true
                            list[#list + 1] = {
                                obj = m,
                                part = hrp,
                                name = m.Name,
                                bossKey = bName,
                                inWorkspace = inWorkspace,
                                fallbackPos = info.pos or hrp.Position,
                            }
                        end
                        break
                    end
                end
            end
        end
    end
    checkFolder(workspace:FindFirstChild("Enemies"), true)
    checkFolder(Svc.ReplicatedStorage, false)
    return list
end

local function nearestFruit(root)
    local best, bestD
    for _, f in ipairs(getFruits()) do
        local d = (f.part.Position - root.Position).Magnitude
        if not bestD or d < bestD then best, bestD = f, d end
    end
    return best
end

                local seen = {}
                for _, item in ipairs(g.collect()) do
                    local pos = item.inWorkspace == false and item.fallbackPos or item.part.Position
                    local dist = (pos - root.Position).Magnitude
                    if dist <= config.espRange then
                        seen[item.obj] = true
                        local tag = g.tags[item.obj]
                        if not tag then
                            tag = createTag(item.part, g.color, g.highlight and item.inWorkspace and item.obj or nil)
                            g.tags[item.obj] = tag

                        tag.txt.Text = string.format("%s [%d]", item.name, math.floor(dist))
                    end
                end
                for obj, tag in pairs(g.tags) do
                    if not seen[obj] then
                        tag.destroy()
                        g.tags[obj] = nil
                    end
                end
            elseif not config[g.key] then
                clearGroup(g)
            end
        end
        task.wait(0.5)

local function normalizeName(str)
    str = tostring(str or ""):lower():gsub("[^a-z]", "")
    str = str:gsub("ies$", "y")
    str = str:gsub("ves$", "f")
    str = str:gsub("men$", "man")
    str = str:gsub("s$", "")
    return str
end

local function isQuestMatching(qTitle, targetMob)
    if not qTitle or qTitle == "" or not targetMob then return false end
    if qTitle:find(targetMob, 1, true) then return true end
    local normTitle = normalizeName(qTitle)
    local normMob = normalizeName(targetMob)
    return normMob ~= "" and normTitle:find(normMob, 1, true) ~= nil
end

local function getBestQuestForLevel()
    CurrentSea = detectSea()
    local lv = playerNumber("Level") or 1
    local list = QuestData[CurrentSea] or QuestData[1]
    local chosen = list[1]
    for _, q in ipairs(list) do
        if lv >= q.lv then
            chosen = q
        else
            break
        end
    end
    return chosen
end

        return true, tostring(t or "")
    end)
    if ok then return visible, titleText end
    return false, ""
end

local function findMobByName(mobName, root, maxRange)
    local folder = workspace:FindFirstChild("Enemies")
    if not folder then return nil end
    local best, bestD = nil, maxRange or math.huge
    for _, m in ipairs(folder:GetChildren()) do
        if isAlive(m) then
            local match = (not mobName) or (m.Name == mobName) or (m.Name:find(mobName, 1, true) ~= nil)
            if match then
                local d = (m.HumanoidRootPart.Position - root.Position).Magnitude
                if d < bestD then
                    best, bestD = m, d
                end
            end
        end
    end
    return best
end

local function applyBringMob(primaryMob, allInRange)
    if not config.bringMob or not isAlive(primaryMob) then return end
    local folder = workspace:FindFirstChild("Enemies")
    if not folder then return end
    local anchorCF = primaryMob.HumanoidRootPart.CFrame
    local anchorPos = anchorCF.Position
    if anchorPos.Y < -5 then return end
    local targetName = primaryMob.Name

    pcall(function()
        if sethiddenproperty then
            sethiddenproperty(LP, "SimulationRadius", math.huge)
        end
    end)

    for _, m in ipairs(folder:GetChildren()) do
        if isAlive(m) and (allInRange or m.Name == targetName) then
            local hrp = m:FindFirstChild("HumanoidRootPart")
            local hum = m:FindFirstChildOfClass("Humanoid")
            if hrp and hum and (hrp.Position - anchorPos).Magnitude <= config.bringRange then
                pcall(function()
                    hrp.CFrame = anchorCF
                    hrp.AssemblyLinearVelocity = Vector3.zero
                    hrp.AssemblyAngularVelocity = Vector3.zero
                    hrp.CanCollide = false
                    hrp.Size = Vector3.new(60, 60, 60)
                    hrp.Transparency = 1
                    hum.WalkSpeed = 0
                    hum.JumpPower = 0
                    if hum:FindFirstChild("Animator") then
                        hum.Animator:Destroy()
                    end
                end)
            end
        end
    end
end

            end
            if config.autoUpgradeV4 then
                invokeCommF("UpgradeRace", "Buy")

        task.wait(0.6)

-- =====================================================
--  VÒNG LẶP DI CHUYỂN CHÍNH (HEARTBEAT - KHÔNG GỌI HÀM YIELD)
-- =====================================================
connect(Svc.RunService.Heartbeat, function(dt)
    if not anyAutoActive() then
        State.target = nil
        State.statusText = "Đang chờ"
        stopFarmBody()
        return
    end
    if os.clock() < State.busyUntil then return end

    local char, hum, root = getChar()
    if not (char and hum and root) or hum.Health <= 0 then
        stopFarmBody()
        return
    end

    -- Chống rơi khi đang farm trên không
    if not config.fly then
        if not State.farmBV or State.farmBV.Parent ~= root then
            stopFarmBody()
            State.farmBV = UI.new("BodyVelocity", {
                MaxForce = Vector3.new(1e5, 1e5, 1e5),
                Velocity = Vector3.zero,
            }, root)
        else
            State.farmBV.Velocity = Vector3.zero
        end
    else
        stopFarmBody()
    end

    local now = os.clock()

    -- 1. AUTO FARM CHEST
    if config.autoChest then
        local bestChest, bestD = nil, math.huge
        for _, c in ipairs(getChests()) do
            if not State.chestBlacklist[c.obj] or now > State.chestBlacklist[c.obj] then
                local d = (c.part.Position - root.Position).Magnitude
                if d < bestD then bestChest, bestD = c, d end
            end
        end
        if bestChest then
            State.statusText = "Nhặt rương: " .. bestChest.name
            moveTowards(root, bestChest.part.Position, bestChest.part.Position, dt)
            if bestD <= 12 then
                if firetouchinterest then
                    pcall(function()
                        firetouchinterest(root, bestChest.part, 0)
                        firetouchinterest(root, bestChest.part, 1)
                    end)
                end
                State.chestBlacklist[bestChest.obj] = now + 5
            end
        else
            State.statusText = "Hết rương trên map"
        end
        return
    end

    -- 2. AUTO RAID ĐẢO (DỌN QUÁI & BAY QUA ĐẢO 1 -> 5)
    if config.autoFarmRaid then
        local raidMob = findMobByName(nil, root, 1500)
        if isAlive(raidMob) then
            State.target = raidMob
            State.statusText = "Raid: Đánh " .. raidMob.Name
            applyBringMob(raidMob, true)
            local mobPos = raidMob.HumanoidRootPart.Position
            moveTowards(root, mobPos + Vector3.new(0, config.farmHeight, config.farmDist), mobPos, dt)
        else
            State.target = nil
            local islPart, islNum = findActiveRaidIsland()
            if islPart then
                State.statusText = "Raid: Bay sang Đảo " .. tostring(islNum)
                moveTowards(root, islPart.Position + Vector3.new(0, 60, 0), islPart.Position, dt)
            elseif config.autoStartRaid then
                local btnPos = (CurrentSea == 2) and Vector3.new(-6438, 250, -4501) or Vector3.new(-5017, 314, -2823)
                State.statusText = "Raid: Đến phòng khởi động"
                moveTowards(root, btnPos, btnPos, dt)
            else
                State.statusText = "Chờ vào phòng Raid..."
            end
        end
        return
    end

    -- 3. AUTO TRIAL V4 & MIRAGE GEAR
    if config.autoMirageGear then
        local mirage = findMirageIsland()
        if mirage then
            local gear = nil
            for _, d in ipairs(mirage:GetDescendants()) do
                if d:IsA("BasePart") and d.Name:lower():find("gear") and d.Transparency < 1 then
                    gear = d
                    break
                end
            end
            if gear then
                State.statusText = "V4: Nhặt Bánh Răng Xanh!"
                moveTowards(root, gear.Position + Vector3.new(0, 2, 0), gear.Position, dt)
            else
                local topPart = findPart(mirage)
                local topPos = topPart and (topPart.Position + Vector3.new(0, 120, 0)) or root.Position
                State.statusText = "V4: Nhìn trăng trên Đảo Bí Ẩn"
                moveTowards(root, topPos, topPos, dt)
            end
        else
            State.statusText = "Chưa thấy Đảo Bí Ẩn trên server"
        end
        return
    end

    if config.autoTrialV4 then
        local race = getPlayerRace()
        local trialMob = findMobByName(nil, root, 800)
        if isAlive(trialMob) then
            State.target = trialMob
            State.statusText = "Trial V4 (" .. race .. "): Đánh " .. trialMob.Name
            applyBringMob(trialMob, true)
            local mobPos = trialMob.HumanoidRootPart.Position
            moveTowards(root, mobPos + Vector3.new(0, config.farmHeight, config.farmDist), mobPos, dt)
        else
            State.target = nil
            local finishPart = nil
            local map = workspace:FindFirstChild("Map")
            if map then
                for _, d in ipairs(map:GetDescendants()) do
                    if d:IsA("BasePart") and (d.Name == "FinishPart" or d.Name == "TrialFinish" or d.Name == "EndPoint") then
                        if (d.Position - root.Position).Magnitude <= 1500 then
                            finishPart = d
                            break
                        end
                    end
                end
            end
            if finishPart then
                State.statusText = "Trial V4: Bay về đích!"
                moveTowards(root, finishPart.Position, finishPart.Position, dt)
            else
                local doorPos = RaceTrialDoors[race] or TempleOfTimePos
                State.statusText = "Trial V4: Đến cửa tộc " .. race
                moveTowards(root, doorPos, doorPos, dt)
            end
        end
        return
    end

    -- 4. AUTO FARM ALL BOSS
    if config.autoFarmAllBoss then
        local bosses = getActiveBosses()
        local chosenBoss = bosses[1]
        if chosenBoss then
            local bq = BossQuests[chosenBoss.bossKey]
            if config.autoBossQuest and bq and bq.qName then
                local hasQ, qTitle = getQuestGuiInfo()
                if not hasQ or not isQuestMatching(qTitle, chosenBoss.bossKey) then
                    State.target = nil
                    State.statusText = "Nhận NV Boss: " .. chosenBoss.bossKey
                    moveTowards(root, bq.npc + Vector3.new(0, 5, 0), bq.npc, dt)
                    return
                end
            end
            if chosenBoss.inWorkspace and isAlive(chosenBoss.obj) then
                State.target = chosenBoss.obj
                State.statusText = "Đánh Boss: " .. chosenBoss.name
                local mobPos = chosenBoss.part.Position
                moveTowards(root, mobPos + Vector3.new(0, config.farmHeight, config.farmDist), mobPos, dt)
            else
                State.target = nil
                State.statusText = "Bay tới bãi Boss: " .. chosenBoss.name
                moveTowards(root, chosenBoss.fallbackPos + Vector3.new(0, 20, 0), chosenBoss.fallbackPos, dt)
            end
        else
            State.target = nil
            State.statusText = "Chưa có Boss sống trên server"
        end
        return
    end

    -- 5. AUTO FARM LEVEL / AUTO MASTERY
    if config.autoFarmLevel or config.autoMastery then
        local q = getBestQuestForLevel()
        State.cachedQuest = q
        if not q then return end

        if config.autoQuest then
            local hasQ, qTitle = getQuestGuiInfo()
            if not (hasQ and isQuestMatching(qTitle, q.mob)) then
                State.target = nil
                State.statusText = "Nhận NV: " .. q.mob
                moveTowards(root, q.npc + Vector3.new(0, 5, 0), q.npc, dt)
                return
            end
        end

        if not isAlive(State.target) or not State.target.Name:find(q.mob, 1, true) then
            if now - State.lastScan >= 0.15 then
                State.lastScan = now
                State.target = findMobByName(q.mob, root, math.huge)
            end
        end

        if isAlive(State.target) then
            State.statusText = (config.autoMastery and "Mastery: " or "Cày NV: ") .. State.target.Name
            applyBringMob(State.target, false)
            local mobPos = State.target.HumanoidRootPart.Position
            moveTowards(root, mobPos + Vector3.new(0, config.farmHeight, config.farmDist), mobPos, dt)
        else
            -- Quét lần lượt các bãi spawn của quái để không bao giờ đứng kẹt 1 chỗ
            local spawns = getMobSpawnPoints(q.mob, q.mobPos)
            if now - State.lastSpawnSwitch >= 2.5 then
                State.lastSpawnSwitch = now
                State.spawnIndex = (State.spawnIndex % #spawns) + 1
            end
            local waitPos = spawns[State.spawnIndex] or q.mobPos
            State.statusText = "Tìm bãi spawn: " .. q.mob
            moveTowards(root, waitPos + Vector3.new(0, math.max(config.farmHeight, 18), 0), waitPos, dt)
        end
        return
    end

    -- 6. SĂN QUÁI THEO TÊN HOẶC CÀY QUÁI GẦN NHẤT
    local huntName = (config.huntOnly and config.huntName) or nil
    if not isAlive(State.target) or (huntName and State.target.Name ~= huntName) then
        if now - State.lastScan >= 0.2 then
            State.lastScan = now
            State.target = findMobByName(huntName, root, huntName and math.huge or config.farmRange)
        end
    end

    if isAlive(State.target) then
        State.statusText = "Đánh: " .. State.target.Name
        applyBringMob(State.target, false)
        local mobPos = State.target.HumanoidRootPart.Position
        moveTowards(root, mobPos + Vector3.new(0, config.farmHeight, config.farmDist), mobPos, dt)
    else
        State.statusText = huntName and ("Chờ: " .. huntName) or "Không có quái gần đây"
    end
end)

task.spawn(function()
    while State.running do
        local char, hum, root = getChar()
        if char and hum and root and hum.Health > 0 then
            local farmingMob = isAlive(State.target) and (
                config.autoFarmLevel or config.autoFarmNear or config.autoFarmAllBoss
                or config.autoMastery or config.autoFarmRaid or config.autoTrialV4
                or (config.huntOnly and config.huntName)
            )
            local nearTarget = false
            local targetPos = nil
            if farmingMob then
                targetPos = State.target.HumanoidRootPart.Position
                local d = (root.Position - targetPos).Magnitude
                nearTarget = d <= math.max(40, config.farmHeight + config.farmDist + 15)
            end

            if config.autoMastery and nearTarget and isAlive(State.target) then
                local mHum = State.target:FindFirstChildOfClass("Humanoid")
                local hpPct = mHum and mHum.MaxHealth > 0 and (mHum.Health / mHum.MaxHealth * 100) or 100
                if hpPct <= config.masteryHP then
                    equipByType(char, hum, config.masteryType)
                    aimCameraAt(targetPos)
                    if config.skillZ then pressSkillKey(Enum.KeyCode.Z) end
                    if config.skillX then pressSkillKey(Enum.KeyCode.X) end
                    if config.skillC then pressSkillKey(Enum.KeyCode.C) end
                    if config.skillV then pressSkillKey(Enum.KeyCode.V) end
                    if config.skillF then pressSkillKey(Enum.KeyCode.F) end
                    performAttack(char)
                else
                    if config.autoEquip then equipByType(char, hum, config.weaponType) end
                    if config.farmAttack then performAttack(char) end
                end
            else
                if config.autoEquip and (config.autoClick or farmingMob) then
                    equipByType(char, hum, config.weaponType)
                end
                if config.autoFarmRaid and config.raidUseSkill and nearTarget and targetPos then
                    aimCameraAt(targetPos)
                    pressSkillKey(Enum.KeyCode.Z)
                    pressSkillKey(Enum.KeyCode.X)
                    pressSkillKey(Enum.KeyCode.C)
                end
                if config.autoClick or (config.farmAttack and nearTarget) then
                    performAttack(char)
                end
            end
        end
        task.wait(1 / math.max(config.clickRate, 1))
    end
end)

                        end

            invokeCommF("Cousin", "Buy")
        end

        if config.autoFruit and os.clock() >= State.busyUntil then
            local _, hum, root = getChar()
            if root and hum and hum.Health > 0 then
                local best = nearestFruit(root)
                if best then
                    local dur = teleportTo(best.part.CFrame)
                    State.busyUntil = os.clock() + dur + 1.5
                    if firetouchinterest then
                        pcall(function()
                            firetouchinterest(root, best.part, 0)
                            firetouchinterest(root, best.part, 1)
                        end)
                    end
                    notify("Nhặt trái", best.name, 2, T.accent)
                end
            end
        end

        if config.fruitAlert then
            local _, _, root = getChar()
            local current = {}
            for _, f in ipairs(getFruits()) do
                current[f.obj] = true
                if not State.knownFruits[f.obj] then
                    State.knownFruits[f.obj] = true
                    local detail = f.name
                    if root then
                        detail = detail .. " - cách " .. tostring(math.floor((f.part.Position - root.Position).Magnitude)) .. " đv"
                    end
                    notify("Có trái ác quỷ!", detail, 6, T.accent, true)
                end
            end
            for obj in pairs(State.knownFruits) do
                if not current[obj] then State.knownFruits[obj] = nil end
            end
        else
            table.clear(State.knownFruits)
        end
        task.wait(1.5)
    end
end)

-- Tự cộng điểm chỉ số
task.spawn(function()
    local lastPts, stuck = nil, 0
    while State.running do
        if config.autoStats then
            local pts = playerNumber("Points")
            if not pts then
                if not State.statWarned then
                    State.statWarned = true
                    notify("Cộng điểm", "Không đọc được số điểm còn lại", 5, T.bad, true)
                end
            elseif pts > 0 then
                if pts == lastPts then stuck = stuck + 1 else stuck = 0 end
                lastPts = pts
                if stuck >= 3 and not State.statWarned then
                    State.statWarned = true
                    notify("Cộng điểm", "Điểm không giảm sau 3 lần thử", 5, T.bad, true)
                end
                invokeCommF("AddPoint", config.statType, math.min(pts, config.statStep))
            end
        else
            lastPts, stuck = nil, 0
        end
        task.wait(1.5)
    end
end)

local function boostFPS()
    pcall(function() settings().Rendering.QualityLevel = Enum.QualityLevel.Level01 end)
    for _, v in ipairs(workspace:GetDescendants()) do
        pcall(function()
            if v:IsA("BasePart") then
                v.Material = Enum.Material.SmoothPlastic
                v.Reflectance = 0
                v.CastShadow = false
            elseif v:IsA("Decal") or v:IsA("Texture") then
                v.Transparency = 1
            elseif v:IsA("ParticleEmitter") or v:IsA("Trail") or v:IsA("Smoke") or v:IsA("Fire") or v:IsA("Sparkles") then
                v.Enabled = false
            end
        end)
    end
    notify("Giảm Lag (FPS Boost)", "Đã tối ưu đồ họa mượt hơn!", 3, T.good, true)
end

local function redeemAllCodes()

local function serverHop()
    notify("Đổi server", "Đang tìm server còn chỗ...", 3)
    task.spawn(function()
        local ok, found = pcall(function()
            local url = ("https://games.roblox.com/v1/games/%d/servers/Public?sortOrder=Asc&limit=100"):format(game.PlaceId)
            local data = Svc.HttpService:JSONDecode(game:HttpGet(url))
            for _, s in ipairs(data.data or {}) do
                if s.id ~= game.JobId and s.playing < s.maxPlayers - 1 then
                    Svc.TeleportService:TeleportToPlaceInstance(game.PlaceId, s.id, LP)
                    return true
                end
            end
            return false
        end)
        if not ok or not found then
            notify("Đổi server", "Không tìm được server phù hợp", 4, T.bad)
        end
    end)
end

local function rejoin()
    notify("Vào lại server", "Đang vào lại server này...", 3)
    local ok = pcall(function()
        Svc.TeleportService:TeleportToPlaceInstance(game.PlaceId, game.JobId, LP)
    end)
    if not ok then Svc.TeleportService:Teleport(game.PlaceId, LP) end
end

    local list, seen = {}, {}

local function commas(v)
    return (tostring(v):reverse():gsub("(%d%d%d)", "%1,"):reverse():gsub("^,", ""))
end

local function playerStat(name)
    local v = playerNumber(name)
    return v and commas(v) or "—"
end

local frames, lastT = 0, os.clock()
connect(Svc.RunService.RenderStepped, function() frames = frames + 1 end)
task.spawn(function()
    while State.running do
        local now = os.clock()
        local fps = math.floor(frames / math.max(now - lastT, 0.001))
        frames, lastT = 0, now

        local okP, ping = pcall(function()
            return Svc.Stats.Network.ServerStatsItem["Data Ping"]:GetValue()
        end)
        local pingStr = okP and (tostring(math.floor(ping)) .. " ms") or "—"
        infoFPS.Text = tostring(fps) .. " FPS • " .. pingStr

        local t = workspace.DistributedGameTime
        infoPlayers.Text = string.format("%d/%d • %02d:%02d",
            #Svc.Players:GetPlayers(), Svc.Players.MaxPlayers, math.floor(t / 60), math.floor(t % 60))

        infoSea.Text = "Sea " .. tostring(CurrentSea)
        infoRace.Text = getPlayerRace()
        infoLevel.Text = playerStat("Level")
        infoBeli.Text = playerStat("Beli")
        infoFrag.Text = playerStat("Fragments")
        infoPoints.Text = playerStat("Points")
        infoStatus.Text = State.statusText

        local bestQ = getBestQuestForLevel()
        infoQuest.Text = bestQ and (bestQ.mob .. " (Lv " .. tostring(bestQ.lv) .. "+)") or "—"
        task.wait(0.5)
    end
end)

State.unload = function()
    State.running = false
    State.target = nil
    stopFly()
    stopFarmBody()
    pcall(function() Svc.RunService:Set3dRenderingEnabled(true) end)

    local _, hum = getChar()
    if hum and config.speedBoost then
        hum.WalkSpeed = State.origWalk
        hum.JumpPower = State.origJump
    end
    restoreNoclip()
    applyFullbright(false)
    for _, g in ipairs(espGroups) do clearGroup(g) end
    if State.currentTween then State.currentTween:Cancel() end

    for _, c in ipairs(State.connections) do c:Disconnect() end
    table.clear(State.connections)
    gui:Destroy()
    if env.BF_Unload == State.unload then env.BF_Unload = nil end
end
